## Step 2. The contrast gate

Create `scripts/check-contrast.mjs`:

```js
// Checks the pairings the design relies on against WCAG 2.2, straight from
// styles/tokens.css, and writes docs/design-system/contrast-report.md.
// Exit code 1 if any pairing is under its bar.
import { readFileSync, writeFileSync } from "node:fs";

const css = readFileSync("styles/tokens.css", "utf8");
const token = (name) => {
  const match = css.match(new RegExp(`--${name}:\\s*(#[0-9a-fA-F]{6})\\b`));
  if (!match) throw new Error(`--${name} is not a six-digit hex in styles/tokens.css`);
  return match[1].toLowerCase();
};
const luminance = (hex) => {
  const [r, g, b] = [1, 3, 5]
    .map((i) => parseInt(hex.slice(i, i + 2), 16) / 255)
    .map((c) => (c <= 0.04045 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4));
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
};
const ratio = (a, b) => {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (hi + 0.05) / (lo + 0.05);
};

// [foreground, background, minimum, what it is used for]
const PAIRS = [
  ["ink", "bg", 7, "running text"],
  ["ink", "surface", 7, "running text on a tone-step section"],
  ["muted", "bg", 4.5, "secondary notes"],
  ["muted", "surface", 4.5, "secondary notes on a tone-step section"],
  ["accent", "bg", 3, "the button against the page"],
  ["accent", "surface", 3, "the button and the price block against a tone-step section"],
  ["accent-ink", "accent", 4.5, "the button label and the price"],
  ["accent-ink", "accent-strong", 4.5, "the button label on hover"],
  ["inv-ink", "inv-bg", 7, "the closing section"],
  ["inv-muted", "inv-bg", 4.5, "the legal line"],
  ["inv-bg", "inv-ink", 4.5, "the ink button's label"],
];

let failed = 0;
const rows = PAIRS.map(([fg, bg, min, use]) => {
  const value = ratio(token(fg), token(bg));
  const ok = value >= min;
  if (!ok) failed++;
  return `| \`--${fg}\` on \`--${bg}\` | ${token(fg)} on ${token(bg)} | ${value.toFixed(2)} | ${min} | ${ok ? "pass" : "**FAIL**"} | ${use} |`;
});

const report = [
  "# Contrast report",
  "",
  "Generated by `npm run check:contrast` from `styles/tokens.css`. Do not edit by hand.",
  "",
  "| Pairing | Colours | Ratio | Minimum | Result | Use |",
  "|---|---|---:|---:|---|---|",
  ...rows,
  "",
].join("\n");
writeFileSync("docs/design-system/contrast-report.md", report);
console.log(report);
if (failed) {
  console.error(`${failed} pairing(s) under the bar.`);
  process.exit(1);
}
```

Run `npm run check:contrast` and commit the report it writes.

In `.github/workflows/ci.yml`, in the `quality` job after "Unit tests", add:

```yaml
      # The pairings the design relies on, checked against WCAG 2.2 straight
      # from styles/tokens.css. The committed report must be current.
      - name: Contrast gate
        run: npm run check:contrast && git diff --exit-code docs/design-system/contrast-report.md
```

## Step 4. The social image

It is a photograph of the real first screen, so it can never drift from it.

Create `scripts/build-og.ts`:

```ts
// Photographs the first screen at 1200 x 630 for link previews.
//   npm run build && npx next start -p 3417      (in another shell)
//   npm run build:og
import { chromium } from "playwright";
import { copyFileSync, writeFileSync } from "node:fs";
import { seo } from "../data/angeronia";

const PORT = Number(process.env.OG_PORT ?? 3417);

async function main() {
  const browser = await chromium.launch();
  const context = await browser.newContext({
    viewport: { width: 1200, height: 630 },
    deviceScaleFactor: 1,
    reducedMotion: "reduce",
  });
  const page = await context.newPage();
  await page.goto(`http://localhost:${PORT}/`, { waitUntil: "networkidle" });
  await page.evaluate(() => document.fonts.ready);
  // The preview shows the name and the sentence. The navigation and the
  // button are controls, and a picture of a control is a lie.
  await page.addStyleTag({ content: "header, main > section:first-child a { visibility: hidden !important; }" });
  await page.waitForTimeout(500);
  await page.screenshot({ path: "app/opengraph-image.png", clip: { x: 0, y: 0, width: 1200, height: 630 } });
  await browser.close();
  copyFileSync("app/opengraph-image.png", "app/twitter-image.png");
  writeFileSync("app/opengraph-image.alt.txt", seo.ogImageAlt + "\n");
  writeFileSync("app/twitter-image.alt.txt", seo.ogImageAlt + "\n");
  console.log("wrote app/opengraph-image.png and app/twitter-image.png");
}

main();
```

Run it against a production server, open the PNG, and check: the name spans
the width and is not cut, the sentence is readable, no navigation, no button.
If the name is cut at 630px tall, the hero's `min-block-size` is doing it:
report it rather than changing the hero. Commit the two PNGs and two alt
files. After `npm run build`, confirm the page's HTML has
`og:image` and `twitter:image` meta tags.
